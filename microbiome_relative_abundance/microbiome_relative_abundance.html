Microbiome relative-abundance composition

Synthetic data

The data below are simulated for visual reconstruction only. Each sample sums to 100% relative abundance.

diet_groups <- c(
  "Psy", "Med", "Jap", "Keto", "Veg", "Pec",
  "LF", "West", "Inu", "Amer", "Asp", "HF"
)

families <- c(
  "Erysipelotrichaceae",
  "Lachnospiraceae",
  "Atopobiaceae",
  "Muribaculaceae",
  "Akkermansiaceae",
  "Bacteroidaceae",
  "Bifidobacteriaceae",
  "Clostridiaceae",
  "Eubacterium_coprostanoligenes_group",
  "Lactobacillaceae",
  "Peptococcaceae",
  "Peptostreptococcaceae",
  "Ruminococcaceae",
  "Oscillospiraceae",
  "Staphylococcaceae",
  "Streptococcaceae",
  "Sutterellaceae",
  "Others"
)

samples_per_group <- 5
sample_table <- expand.grid(
  group = diet_groups,
  replicate = seq_len(samples_per_group),
  KEEP.OUT.ATTRS = FALSE,
  stringsAsFactors = FALSE
) |>
  mutate(
    group = factor(group, levels = diet_groups)
  ) |>
  arrange(group, replicate) |>
  mutate(
    sample_index = row_number(),
    ici_score = scales::rescale(sample_index, to = c(0.08, 0.96)),
    sample = sprintf("S%02d", sample_index)
  )

base_shape <- c(
  6.0, 4.1, 2.2, 3.4, 1.1, 1.4, 1.6, 1.2, 0.8,
  1.7, 1.0, 1.4, 1.1, 1.7, 0.8, 0.7, 1.0, 1.5
)

group_effect <- matrix(
  exp(rnorm(length(diet_groups) * length(families), 0, 0.38)),
  nrow = length(diet_groups),
  dimnames = list(diet_groups, families)
)

# Add broad group-level ecological signatures.
group_effect["Keto", c("Lachnospiraceae", "Atopobiaceae")] <- c(3.5, 2.3)
group_effect["Veg", c("Lachnospiraceae", "Muribaculaceae")] <- c(2.4, 2.0)
group_effect["Pec", c("Lactobacillaceae", "Bifidobacteriaceae")] <- c(3.0, 2.1)
group_effect["LF", c("Erysipelotrichaceae", "Bacteroidaceae")] <- c(2.7, 2.0)
group_effect["West", c("Peptostreptococcaceae", "Ruminococcaceae")] <- c(2.6, 2.1)
group_effect["Inu", c("Bifidobacteriaceae", "Lactobacillaceae")] <- c(3.3, 2.1)
group_effect["Amer", c("Muribaculaceae", "Akkermansiaceae")] <- c(2.2, 2.0)
group_effect["Asp", c("Erysipelotrichaceae", "Lactobacillaceae")] <- c(2.5, 2.7)
group_effect["HF", c("Streptococcaceae", "Oscillospiraceae")] <- c(3.2, 2.0)

composition_list <- lapply(seq_len(nrow(sample_table)), function(i) {
  group_name <- as.character(sample_table$group[i])
  score <- sample_table$ici_score[i]

  ici_effect <- rep(1, length(families))
  names(ici_effect) <- families
  ici_effect[c("Lactobacillaceae", "Bifidobacteriaceae", "Oscillospiraceae")] <-
    0.70 + 1.45 * score
  ici_effect[c("Erysipelotrichaceae", "Lachnospiraceae")] <-
    1.15 - 0.35 * score

  shape <- pmax(
    0.24,
    base_shape * group_effect[group_name, ] * ici_effect *
      exp(rnorm(length(families), 0, 0.24))
  )

  abundance <- rgamma(length(families), shape = shape, rate = 1)
  abundance <- 100 * abundance / sum(abundance)

  data.frame(
    sample = sample_table$sample[i],
    sample_index = sample_table$sample_index[i],
    group = group_name,
    replicate = sample_table$replicate[i],
    ici_score = score,
    family = families,
    relative_abundance = abundance
  )
})

abundance_data <- bind_rows(composition_list) |>
  mutate(
    family = factor(family, levels = families),
    group = factor(group, levels = diet_groups)
  )

group_midpoints <- sample_table |>
  group_by(group) |>
  summarise(midpoint = mean(sample_index), .groups = "drop")

write.csv(
  abundance_data,
  "synthetic_microbiome_abundance.csv",
  row.names = FALSE
)
