Map construction

crs_europe <- 3035

world <- ne_countries(scale = "medium", returnclass = "sf")

europe <- world %>%
  filter(continent == "Europe" | iso_a2 %in% c("TR", "CY")) %>%
  left_join(country_data %>% select(iso_a2, carbon_class), by = "iso_a2") %>%
  st_transform(crs_europe)

project_points <- function(data, lon_col, lat_col, x_name, y_name) {
  pts <- st_as_sf(
    data,
    coords = c(lon_col, lat_col),
    crs = 4326,
    remove = FALSE
  ) %>%
    st_transform(crs_europe)
  xy <- st_coordinates(pts)
  data[[x_name]] <- xy[, 1]
  data[[y_name]] <- xy[, 2]
  data
}

make_pies <- function(data, radius = 88000, points_per_slice = 36) {
  classes <- c("Inland marshes", "Peatbogs", "Salt marshes", "Moors and heathland")
  value_cols <- c("inland", "peat", "salt", "moor")
  pieces <- vector("list", nrow(data) * length(classes))
  k <- 1

  for (i in seq_len(nrow(data))) {
    values <- as.numeric(data[i, value_cols])
    angles <- c(0, cumsum(values / sum(values)) * 2 * pi)

    for (j in seq_along(classes)) {
      theta <- seq(angles[j], angles[j + 1], length.out = points_per_slice)
      pieces[[k]] <- tibble(
        iso_a2 = data$iso_a2[i],
        wetland_class = classes[j],
        piece = paste(data$iso_a2[i], j, sep = "_"),
        x = c(data$pie_x[i], data$pie_x[i] + radius * cos(theta), data$pie_x[i]),
        y = c(data$pie_y[i], data$pie_y[i] + radius * sin(theta), data$pie_y[i])
      )
      k <- k + 1
    }
  }

  bind_rows(pieces)
}
